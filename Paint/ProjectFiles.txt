#include "touchpad.h"

static uint16_t brush_color = TOUCH_DEFAULT_COLOR;  // С†РІРµС‚ С‚РµРєСѓС‰РµР№ РєРёСЃС‚Рё
static uint8_t brush_size = TOUCH_DEFAULT_SIZE;     // СЂР°Р·РјРµСЂ С‚РµРєСѓС‰РµР№ РєРёСЃС‚Рё

void touch_set_color(uint16_t color)
{
    brush_color = color;
}

    return brush_color;
}

void touch_set_brush_size(uint8_t size)
{
    if (size < TOUCH_MIN_BRUSH_SIZE) {
        size = TOUCH_MIN_BRUSH_SIZE;
    }

    if (size > TOUCH_MAX_BRUSH_SIZE) {

/* Private functions ---------------------------------------------------------*/
/**
 * @brief  Р’С‹Р±РѕСЂ С‚Р°С‡РїР°РґР° РЅР° С€РёРЅРµ SPI 
 */

{
    HAL_GPIO_WritePin(GPIO_CS, SLAVE_CS_ALL, __LOW); // СЃР±СЂРѕСЃ РІСЃРµС… CS
    HAL_GPIO_WritePin(GPIO_CS, SLAVE_CS1, __HIGH);   // РІС‹Р±РѕСЂ С‚Р°С‡РїР°РґР° 
}

/**
 * @brief  РЎРЅСЏС‚РёРµ РІС‹Р±РѕСЂР° С‚Р°С‡РїР°РґР°

    HAL_GPIO_WritePin(GPIO_CS, SLAVE_CS_ALL, __LOW);

/**
 * @brief         РћРіСЂР°РЅРёС‡РµРЅРёРµ РєРѕРѕСЂРґРёРЅР°С‚ РіСЂР°РЅРёС†Р°РјРё СЌРєСЂР°РЅР°
 * @param  x      Р“РѕСЂРёР·РѕРЅС‚Р°Р»СЊРЅР°СЏ РєРѕРѕСЂРґРёРЅР°С‚Р° 
 * @param  y      Р’РµСЂС‚РёРєР°Р»СЊРЅР°СЏ РєРѕРѕСЂРґРёРЅР°С‚Р° 
 */
void touch_clamp_screen(int *x, int *y)
{
    if (*x < 0) {
        *x = 0;                   // Р»РµРІР°СЏ РіСЂР°РЅРёС†Р°
    }
    if (*y < 0) {
        *y = 0;                   // РІРµСЂС…РЅСЏСЏ РіСЂР°РЅРёС†Р°
    }
    if (*x >= TOUCH_SCREEN_H) {
        *x = TOUCH_SCREEN_H - 1;  // РїСЂР°РІР°СЏ РіСЂР°РЅРёС†Р°
    }
    if (*y >= TOUCH_SCREEN_W) {
        *y = TOUCH_SCREEN_W - 1;  // РЅРёР¶РЅСЏСЏ РіСЂР°РЅРёС†Р°
    }
}

/**
 * @brief         РћС‚СЂРёСЃРѕРІРєР° С‚РѕС‡РєРё РєР°СЃР°РЅРёСЏ 
 * @param  x      Р“РѕСЂРёР·РѕРЅС‚Р°Р»СЊРЅР°СЏ РєРѕРѕСЂРґРёРЅР°С‚Р° 
 * @param  y      Р’РµСЂС‚РёРєР°Р»СЊРЅР°СЏ РєРѕРѕСЂРґРёРЅР°С‚Р° 
 */
void touch_draw_stamp(int x, int y)
{
    int size;
    int left;
    int top;

    size = brush_size;

    left = x - size / 2 + 15;
    top  = y - size / 2;

    touch_clamp_screen(&left, &top);

    draw_rect(size,
              size,
              top,
              left,
              brush_color);
}
/**
 * @brief         Р›РёРЅРёСЏ РјРµР¶РґСѓ РґРІСѓРјСЏ С‚РѕС‡РєР°РјРё (Р°Р»РіРѕСЂРёС‚Рј Р‘СЂРµР·РµРЅС…СЌРјР°)
 * @param  x0     РќР°С‡Р°Р»Рѕ, РіРѕСЂРёР·РѕРЅС‚Р°Р»СЊ
 * @param  y0     РќР°С‡Р°Р»Рѕ, РІРµСЂС‚РёРєР°Р»СЊ
 * @param  x1     РљРѕРЅРµС†, РіРѕСЂРёР·РѕРЅС‚Р°Р»СЊ
 * @param  y1     РљРѕРЅРµС†, РІРµСЂС‚РёРєР°Р»СЊ
 */
void touch_stamp_line(int x0, int y0, int x1, int y1)
{
    int dx;
    int dy;
    int sx;
    int sy;
    int err;
    int e2;
    uint16_t steps = 0;

    dx = ABS(x1 - x0);
    sx = (x0 < x1) ? 1 : -1;
    dy = -ABS(y1 - y0);
    sy = (y0 < y1) ? 1 : -1;
    err = dx + dy;

    while (1) {
        touch_draw_stamp(x0, y0);
        if (x0 == x1 && y0 == y1) {
            break;
        }
        e2 = 2 * err;
        if (e2 >= dy) {
            err += dy;
            x0 += sx;
        }
        if (e2 <= dx) {
            err += dx;
            y0 += sy;
        }
        if (++steps > 600) {
            break;
        }
    }
}

/**
 * @brief         РЈСЃСЂРµРґРЅС‘РЅРЅРѕРµ С‡С‚РµРЅРёРµ РїР°СЂС‹ Y/X 
 * @param  raw_x  РЈСЃСЂРµРґРЅС‘РЅРЅРѕРµ СЃС‹СЂРѕРµ Р·РЅР°С‡РµРЅРёРµ РѕСЃРё X
 * @param  raw_y  РЈСЃСЂРµРґРЅС‘РЅРЅРѕРµ СЃС‹СЂРѕРµ Р·РЅР°С‡РµРЅРёРµ РѕСЃРё Y
 * @return        1 вЂ” РєР°СЃР°РЅРёРµ РµСЃС‚СЊ, 0 вЂ” РЅРµС‚ РІР°Р»РёРґРЅС‹С… Р·РЅР°С‡РµРЅРёР№
 */
int touch_read_averaged(int *raw_x, int *raw_y)
{
    int32_t sum_x = 0;    
    int32_t sum_y = 0;    
    uint8_t valid = 0;

    touch_select();

    for (uint8_t i = 0; i < TOUCH_FILTER_SAMPLES; i++) {
        int vy = read_coordinate(READ_Y);  // СЃРЅР°С‡Р°Р»Р° С‡РёС‚Р°РµРј Y
        int vx = read_coordinate(READ_X);  // Р·Р°С‚РµРј X

        if (vx > 0 && vy > 0) {
            sum_x += vx;
            sum_y += vy;
            valid++;
        }
        HAL_DelayMs(TOUCH_FILTER_DELAY);  // РїР°СѓР·Р° РјРµР¶РґСѓ РёР·РјРµСЂРµРЅРёСЏРјРё
    }

    touch_unselect();

    if (valid == 0) {
        return 0;  // РЅРµС‚ РєР°СЃР°РЅРёСЏ 
    }

    *raw_x = (int)(sum_x / valid);  // СЃСЂРµРґРЅРµРµ РїРѕ X
    *raw_y = (int)(sum_y / valid);  // СЃСЂРµРґРЅРµРµ РїРѕ Y
    return 1;
}

/**
 * @brief         РџСЂРµРѕР±СЂР°Р·РѕРІР°РЅРёРµ РєРѕРѕСЂРґРёРЅР°С‚ РІ РїРёРєСЃРµР»Рё СЌРєСЂР°РЅР°
 * @param  raw_x  РЎС‹СЂРѕРµ Р·РЅР°С‡РµРЅРёРµ РѕСЃРё X С‚Р°С‡РїР°РґР°
 * @param  raw_y  РЎС‹СЂРѕРµ Р·РЅР°С‡РµРЅРёРµ РѕСЃРё Y С‚Р°С‡РїР°РґР°
 * @param  sx     Р“РѕСЂРёР·РѕРЅС‚Р°Р»СЊРЅР°СЏ РєРѕРѕСЂРґРёРЅР°С‚Р° РЅР° РґРёСЃРїР»РµРµ
 * @param  sy     Р’РµСЂС‚РёРєР°Р»СЊРЅР°СЏ РєРѕРѕСЂРґРёРЅР°С‚Р° РЅР° РґРёСЃРїР»РµРµ
 * @return        1 вЂ” РєРѕРѕСЂРґРёРЅР°С‚С‹ РїРѕР»СѓС‡РµРЅС‹, 0 вЂ” РєР°СЃР°РЅРёСЏ РЅРµС‚
 */
int touch_raw_to_screen(int raw_x, int raw_y, int *sx, int *sy)
{
    int rx = raw_x;     int span_x;        
    int ry = raw_y;     int span_y;

    if (rx <= 0 || ry <= 0) {
        return 0;  // РІС‹С…РѕРґ, РµСЃР»Рё РЅРµС‚ РєР°СЃР°РЅРёСЏ
    }

    /* РѕРіСЂР°РЅРёС‡РµРЅРёРµ Р·РЅР°С‡РµРЅРёР№ РґРѕРїСѓСЃС‚РёРјС‹Рј РґРёР°РїР°Р·РѕРЅРѕРј */
    if (rx < TOUCH_MIN_RAW_X) {
        rx = TOUCH_MIN_RAW_X;
    }
    if (rx > TOUCH_MAX_RAW) {
        rx = TOUCH_MAX_RAW;
    }
    if (ry < TOUCH_MIN_RAW_Y) {
        ry = TOUCH_MIN_RAW_Y;
    }
    if (ry > TOUCH_MAX_RAW) {
        ry = TOUCH_MAX_RAW;
    }

    span_x = TOUCH_MAX_RAW - TOUCH_MIN_RAW_X;  // РґРёР°РїР°Р·РѕРЅ РїРѕ X
    span_y = TOUCH_MAX_RAW - TOUCH_MIN_RAW_Y;  // РґРёР°РїР°Р·РѕРЅ РїРѕ Y
    if (span_x <= 0 || span_y <= 0) {
        return 0;
    }

    /* x  - РіРѕСЂРёР·РѕРЅС‚Р°Р»СЊ, y - РІРµСЂС‚РёРєР°Р»СЊ */
    *sx = (rx - TOUCH_MIN_RAW_X) * (TOUCH_SCREEN_H - 1) / span_x;
    *sy = (TOUCH_SCREEN_W - 1) - (ry - TOUCH_MIN_RAW_Y) * (TOUCH_SCREEN_W - 1) / span_y;

    return 1;
}

/**
 * @brief           Р§С‚РµРЅРёРµ РѕРґРЅРѕР№ РєРѕРѕСЂРґРёРЅР°С‚С‹ С‚Р°С‡РїР°РґР° РїРѕ SPI
 * @param  command  РљРѕРјР°РЅРґР° С‡С‚РµРЅРёСЏ: READ_X (0xD0) РёР»Рё READ_Y (0x90)
 * @return          12-Р±РёС‚РЅРѕРµ Р·РЅР°С‡РµРЅРёРµ РєРѕРѕСЂРґРёРЅР°С‚ СЃ С‚Р°С‡РїР°РґР°, РёРЅР°С‡Рµ -1 РїСЂРё РѕС€РёР±РєРµ SPI
 */
int read_coordinate(unsigned char command)
{
    int16_t value;
    // РџРѕРґРіРѕС‚РѕРІРєР° РїР°РєРµС‚Р° (РљРѕРјР°РЅРґР° + 2 Р±Р°Р№С‚Р° РґР»СЏ РїСЂРёРµРјР°)
    uint8_t master_output[] = {command, 0x00, 0x00};  // РўР°С‡РїР°Рґ РѕС‚РїСЂР°РІР»СЏРµС‚ 12-Р±РёС‚РЅС‹Рµ РєРѕРѕСЂРґРёРЅРІР°С‚С‹, СѓРїР°РєРѕРІР°РЅРЅС‹Рµ РІ 2 Р±Р°Р№С‚Р°
    uint8_t master_input[3] = {0}; // РРЅРёС†РёР°Р»РёР·РёСЂСѓРµРј РЅСѓР»СЏРјРё
    
    // РћР±РјРµРЅ РґР°РЅРЅС‹РјРё РїРѕ SPI (Р‘Р»РѕРєРёСЂСѓСЋС‰РёР№ СЂРµР¶РёРј)
    HAL_StatusTypeDef SPI_Status = HAL_SPI_Exchange(&hspi1, master_output, master_input,
                                                    sizeof(master_output),
                                                    SPI_TIMEOUT_DEFAULT);  // РѕС‚РїСЂР°РІРєР° РєРѕРјР°РЅРґС‹

    if (SPI_Status != HAL_OK) {
        xprintf("SPI_Error %d, OVR %d, MODF %d\n", SPI_Status,
                hspi1.ErrorCode & HAL_SPI_ERROR_OVR,
                hspi1.ErrorCode & HAL_SPI_ERROR_MODF);
        HAL_SPI_ClearError(&hspi1);
        return -1;  // РѕС€РёР±РєР° РѕР±РјРµРЅР°
    }

    /*РљРѕРјР±РёРЅРёСЂРѕРІР°РЅРёРµ 12 Р±РёС‚ РґР°РЅРЅС‹С…

    Р‘Р°Р№С‚С‹ 1 Рё 2 СЃРѕРґРµСЂР¶Р°С‚ 16 Р±РёС‚, РёР· РєРѕС‚РѕСЂС‹С… 12 Р±РёС‚ - СЌС‚Рѕ РїРѕР»РµР·РЅР°СЏ РЅР°РіСЂСѓР·РєР° (РєРѕРѕСЂРґРёРЅР°С‚Р°).
    РћР±С‹С‡РЅРѕ 4 РјР»Р°РґС€РёС… Р±РёС‚Р° РїРµСЂРІРѕРіРѕ Р±Р°Р№С‚Р° Р±РµСЃРїРѕР»РµР·РЅС‹, Р° 4 СЃС‚Р°СЂС€РёС… Р±РёС‚Р° РІС‚РѕСЂРѕРіРѕ Р±Р°Р№С‚Р° Р±РµСЃРїРѕР»РµР·РЅС‹.
    
    РЎРѕР±РёСЂР°РµРј 12 Р±РёС‚: (master_input[1] << 4) | (master_input[2] >> 4)

    value = ((master_input[1] & 0x0F) << 8) | master_input[2]; // РџСЂРѕСЃС‚Р°СЏ СЃР±РѕСЂРєР° 16 Р±РёС‚

    РЎС‚Р°СЂС€РёР№ Р±Р°Р№С‚ (master_input[1]) СЃРѕРґРµСЂР¶РёС‚ 4 СЃС‚Р°СЂС€РёС… Р±РёС‚Р°
    РњР»Р°РґС€РёР№ Р±Р°Р№С‚ (master_input[2]) СЃРѕРґРµСЂР¶РёС‚ 8 РјР»Р°РґС€РёС… Р±РёС‚*/

    value = (master_input[1] << 4) | (master_input[2] >> 4);

    return value;
}

/**
 * @brief           РЈСЃСЂРµРґРЅС‘РЅРЅРѕРµ С‡С‚РµРЅРёРµ РѕРґРЅРѕР№ РѕСЃРё (РґР»СЏ button.c)
 * @param  command  РљРѕРјР°РЅРґР° С‡С‚РµРЅРёСЏ: READ_X РёР»Рё READ_Y
 * @param  samples  Р§РёСЃР»Рѕ Р·Р°РјРµСЂРѕРІ РґР»СЏ СѓСЃСЂРµРґРЅРµРЅРёСЏ
 * @return          РЎСЂРµРґРЅРµРµ Р·РЅР°С‡РµРЅРёРµ РєРѕРѕСЂРґРёРЅР°С‚С‹, РёРЅР°С‡Рµ -1
 */
int read_filtered_coordinate(uint8_t command, uint8_t samples)
{
    int32_t sum = 0;
    uint8_t valid = 0;

    for (uint8_t i = 0; i < samples; i++) {
        int val = read_coordinate(command);
        if (val > 0) {   // С‚РѕР»СЊРєРѕ РєРѕСЂСЂРµРєС‚РЅС‹Рµ РёР·РјРµСЂРµРЅРёСЏ 
            sum += val;
            valid++;
        }
        HAL_DelayMs(2);  // РїР°СѓР·Р° РјРµР¶РґСѓ РёР·РјРµСЂРµРЅРёСЏРјРё
    }

    if (valid == 0) {
        return -1;       // СЃРѕРѕС‚РІРµС‚СЃС‚РІРµРЅРЅРѕ РЅРµС‚ РґР°РЅРЅС‹С…
    }

    return sum / valid;  // СЃСЂРµРґРЅРµРµ Р·РЅР°С‡РµРЅРёРµ
}

/**
 * @brief          Р›РёРЅРµР№РЅРѕРµ РїСЂРµРѕР±СЂР°Р·РѕРІР°РЅРёРµ РєРѕРѕСЂРґРёРЅР°С‚С‹ РІ РґСЂСѓРіРѕР№ РґРёР°РїР°Р·РѕРЅ
 * @param  value   РСЃС…РѕРґРЅРѕРµ Р·РЅР°С‡РµРЅРёРµ
 * @param  r1_max  Р’РµСЂС…РЅСЏСЏ РіСЂР°РЅРёС†Р° РёСЃС…РѕРґРЅРѕРіРѕ РґРёР°РїР°Р·РѕРЅР°
 * @param  r1_min  РќРёР¶РЅСЏСЏ РіСЂР°РЅРёС†Р° РёСЃС…РѕРґРЅРѕРіРѕ РґРёР°РїР°Р·РѕРЅР°
 * @param  r2_max  Р’РµСЂС…РЅСЏСЏ РіСЂР°РЅРёС†Р° С†РµР»РµРІРѕРіРѕ РґРёР°РїР°Р·РѕРЅР°
 * @param  r2_min  РќРёР¶РЅСЏСЏ РіСЂР°РЅРёС†Р° С†РµР»РµРІРѕРіРѕ РґРёР°РїР°Р·РѕРЅР°
 * @return         РџСЂРµРѕР±СЂР°Р·РѕРІР°РЅРЅРѕРµ Р·РЅР°С‡РµРЅРёРµ СЃ РѕРіСЂР°РЅРёС‡РµРЅРёРµРј РїРѕ РіСЂР°РЅРёС†Р°Рј
 */
uint16_t transform(uint16_t value, uint16_t r1_max, uint16_t r1_min,
                   uint16_t r2_max, uint16_t r2_min)
{
    int scale;    
    int res;

    if (value <= r1_min) {
        return r2_min;  
    }
    if (value >= r1_max) {
        return r2_max; 
    }

    scale = (r2_max - r2_min) * 1000 / (r1_max - r1_min);  
    res = (value - r1_min) * scale / 1000;  // РїСЂРµРѕР±СЂР°Р·РѕРІР°РЅРёРµ РєРѕРѕСЂРґРёРЅР°С‚ СЃ С‚Р°С‡РїР°РґР° РІ РєРѕРѕСЂРґРёРЅР°С‚С‹ СЌРєСЂР°РЅР°

    if (res < r2_min) {
        res = r2_min;  
    }
    if (res > r2_max) {
        res = r2_max; 
    }

    return res;
}

/**
 * @brief  Р РёСЃРѕРІР°РЅРёРµ Р»РёРЅРёР№ РїСЂРё РґРІРёР¶РµРЅРёРё СЃС‚РёР»СѓСЃР° РїРѕ СЌРєСЂР°РЅСѓ
 */

void touch_draw_lines(void)
{
    int x;
    int y;
    int raw_x;
    int raw_y;

    static int last_x = -1;
    static int last_y = -1;

    //Р§РёС‚Р°РµРј Рё С„РёР»СЊС‚СЂСѓРµРј РєРѕРѕСЂРґРёРЅР°С‚С‹. Р•СЃР»Рё РєР°СЃР°РЅРёСЏ РЅРµС‚, С‚Рѕ С‚РµРєСѓС‰РёР№ С€С‚СЂРёС… Р·Р°РєРѕРЅС‡РёР»СЃСЏ
    if (!touch_read_averaged(&raw_x, &raw_y) || !touch_raw_to_screen(raw_x, raw_y, &x, &y))
    {
        last_x = -1;
        last_y = -1;
        return;
    }

    //Р•СЃР»Рё РїСЂРµРґС‹РґСѓС‰РµР№ С‚РѕС‡РєРё РЅРµС‚, СЌС‚Рѕ РЅР°С‡Р°Р»Рѕ РЅРѕРІРѕРіРѕ С€С‚СЂРёС…Р°.
    if (last_x < 0 || last_y < 0)
    {
        touch_draw_stamp(x, y);

        last_x = x;
        last_y = y;

        return;
    }

    //Р’С‹С‡РёСЃР»СЏРµРј РїРµСЂРµРјРµС‰РµРЅРёРµ РїРµСЂР°
    int dx = x - last_x;
    int dy = y - last_y;

    //Р—Р°С‰РёС‚Р° РѕС‚ СЃРєР°С‡РєР° РєРѕРѕСЂРґРёРЅР°С‚. Р•СЃР»Рё РјРµР¶РґСѓ РґРІСѓРјСЏ РёР·РјРµСЂРµРЅРёСЏРјРё РєРѕРѕСЂРґРёРЅР°С‚Р° РёР·РјРµРЅРёР»Р°СЃСЊ СЃР»РёС€РєРѕРј СЃРёР»СЊРЅРѕ, РЅРµ СЃРѕРµРґРёРЅСЏРµРј С‚РѕС‡РєРё Р»РёРЅРёРµР№.
    if ((dx * dx + dy * dy) > TOUCH_MAX_JUMP_SQ)
    {
        // РЎС‡РёС‚Р°РµРј С‚РµРєСѓС‰СѓСЋ С‚РѕС‡РєСѓ РЅР°С‡Р°Р»РѕРј РїСЂРѕРґРѕР»Р¶РµРЅРёСЏ С€С‚СЂРёС…Р°
        touch_draw_stamp(x, y);

        last_x = x;
        last_y = y;

        return;
    }
    touch_stamp_line(last_x, last_y, x, y); // РќРѕСЂРјР°Р»СЊРЅРѕРµ РїСЂРѕРґРѕР»Р¶РµРЅРёРµ С€С‚СЂРёС…Р°

    //РўРµРєСѓС‰Р°СЏ С‚РѕС‡РєР° СЃС‚Р°РЅРѕРІРёС‚СЃСЏ РїСЂРµРґС‹РґСѓС‰РµР№ РґР»СЏ СЃР»РµРґСѓСЋС‰РµРіРѕ РІС‹Р·РѕРІР°.
    last_x = x;
    last_y = y;
}

/**
 * @brief  РћР±СЂР°Р±РѕС‚РєР° РєР°СЃР°РЅРёСЏ Рё РѕС‚СЂРёСЃРѕРІРєР° РјР°СЂРєРµСЂР° 
 *
 * Р РёСЃСѓРµС‚ РјР°СЂРєРµСЂ (РїСЂСЏРјРѕСѓРіРѕР»СЊРЅРёРє 10Г—10) РІ С‚РѕС‡РєРµ РєР°СЃР°РЅРёСЏ, РµСЃР»Рё РєРѕРѕСЂРґРёРЅР°С‚С‹
 * РёР·РјРµРЅРёР»РёСЃСЊ Р±РѕР»СЊС€Рµ С‡РµРј РЅР° 20 РїРёРєСЃРµР»РµР№
 */
void process_touchpad(void)
{
    int raw_x;    int sx;    static int last_raw_x = 0;
    int raw_y;    int sy;    static int last_raw_y = 0;

    if (!touch_read_averaged(&raw_x, &raw_y) ||
        !touch_raw_to_screen(raw_x, raw_y, &sx, &sy)) {
        return;  // РІС‹С…РѕРґ, РµСЃР»Рё РЅРµС‚ РєР°СЃР°РЅРёСЏ
    }

    /* РёРіРЅРѕСЂРёСЂРѕРІР°С‚СЊ "РґСЂРѕР¶Р°РЅРёРµ" РїСЂРё РєР°СЃР°РЅРёРё*/
    if (ABS(raw_x - last_raw_x) < TOUCH_NOISE_THRESHOLD &&
        ABS(raw_y - last_raw_y) < TOUCH_NOISE_THRESHOLD) {
        return;
    }

    last_raw_x = raw_x;
    last_raw_y = raw_y;

    /* РјР°СЂРєРµСЂ 10Г—10 */
    draw_rect(10, 10, sy, sx, TOUCH_DEFAULT_COLOR);
}

    Lcd_Write_Data(x2 & 0xFF);

/**

 * @param  length  Р’С‹СЃРѕС‚Р° РІ РїРёРєСЃРµР»СЏС…
 * @param  width   РЁРёСЂРёРЅР° РІ РїРёРєСЃРµР»СЏС…
 * @param  y       РљРѕРѕСЂРґРёРЅР°С‚Р° d_l
 * @param  x       РљРѕРѕСЂРґРёРЅР°С‚Р° d_w
 * @param  color   Р¦РІРµС‚ Р·Р°Р»РёРІРєРё (RGB565)
 */
void draw_rect(uint16_t length, uint16_t width, uint16_t y, uint16_t x, uint16_t color)
{
    uint8_t r = (color >> 8) & 0xF8;
    uint8_t g = (color >> 3) & 0xFC;
    uint8_t b = (color << 3);
    uint32_t pixels = (uint32_t)length * width;

    Lcd_select();
    HAL_GPIO_WritePin(GPIO_0, RS, 0);
    Address_set(x, y, x + width - 1, y + length - 1);
    HAL_GPIO_WritePin(GPIO_0, RS, 1);

    __HAL_SPI_ENABLE(&hspi0);

    while (pixels--) {
        while (!(hspi0.Instance->INT_STATUS & SPI_INT_STATUS_TX_FIFO_NOT_FULL_M));
        hspi0.Instance->TXDATA = r;
        hspi0.Instance->TXDATA = g;
        hspi0.Instance->TXDATA = b;
    }

    __HAL_SPI_DISABLE(&hspi0);
    Lcd_unselect();
}

/**
 * @brief          Р—Р°Р»РёРІРєР° РїСЂСЏРјРѕСѓРіРѕР»СЊРЅРѕР№ РѕР±Р»Р°СЃС‚Рё С‡РµСЂРµР· DMA
 * @param  length  Р’С‹СЃРѕС‚Р° РІ РїРёРєСЃРµР»СЏС…
 * @param  width   РЁРёСЂРёРЅР° РІ РїРёРєСЃРµР»СЏС…
 * @param  d_l     РЎРјРµС‰РµРЅРёРµ РїРѕ РІРµСЂС‚РёРєР°Р»Рё
 * @param  d_w     РЎРјРµС‰РµРЅРёРµ РїРѕ РіРѕСЂРёР·РѕРЅС‚Р°Р»Рё
 * @param  color   Р¦РІРµС‚ Р·Р°Р»РёРІРєРё (RGB565)
 */
void ClearMassDMA(uint16_t length, uint16_t width, uint16_t d_l, uint16_t d_w, uint16_t color)
{
    uint8_t red   = (color >> 8) & 0xF8;
    uint8_t green = (color >> 3) & 0xFC;
    uint8_t blue  = (color << 3);

    Lcd_select();
    length++;
    width++;
    Address_set(d_w, d_l, width - 2 + d_w, length - 2 + d_l);
    HAL_GPIO_WritePin(GPIO_0, RS, 1);
    hspi0.ErrorCode = HAL_SPI_ERROR_NONE;
    hspi0.Instance->TX_THR = 1;

    if (!(hspi0.Instance->ENABLE & SPI_ENABLE_M)) {
        __HAL_SPI_ENABLE(&hspi0);
    }

    for (uint16_t m = 0; m < length; m++) {
        for (uint16_t i = 0; i < width; i++) {
            DMA_BUF[i * 3 + 2] = red;
            DMA_BUF[i * 3 + 0] = green;
            DMA_BUF[i * 3 + 1] = blue;
        }
        lcd_dma_send_row(width);
    }
    lcd_dma_end();
}
